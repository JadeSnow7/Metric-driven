import unittest

from pricing import apply_discount


class ApplyDiscountTests(unittest.TestCase):
    def test_no_discount(self) -> None:
        self.assertEqual(apply_discount(10.0, 0), 10.0)

    def test_regular_discount(self) -> None:
        self.assertEqual(apply_discount(19.99, 0.1), 17.99)

    def test_half_cent_rounds_up(self) -> None:
        self.assertEqual(apply_discount(0.25, 0.5), 0.13)

    def test_half_cent_rounds_up_when_float_is_slightly_below(self) -> None:
        # 2.01 * 0.5 is stored as 1.00499999... in binary floating point.
        self.assertEqual(apply_discount(2.01, 0.5), 1.01)

    def test_below_half_cent_rounds_down(self) -> None:
        self.assertEqual(apply_discount(0.2498, 0.5), 0.12)

    def test_very_large_and_infinite_prices(self) -> None:
        self.assertEqual(apply_discount(1e30, 0.5), 5e29)
        self.assertEqual(apply_discount(float("inf"), 0.5), float("inf"))

    def test_full_discount(self) -> None:
        self.assertEqual(apply_discount(5.0, 1), 0.0)

    def test_invalid_rate(self) -> None:
        with self.assertRaises(ValueError):
            apply_discount(5.0, 1.5)


if __name__ == "__main__":
    unittest.main()

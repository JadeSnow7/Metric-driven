   1 use rein_ch01_helloworld::rein::{run_agent_loop, OpenAiHttp, ToolError, Workspace};
   2 use std::{
   3     future::Future,
   4     pin::Pin,
   5     sync::{Arc, Mutex},
   6 };
   7 
   8 struct Replay {
   9     responses: Arc<Mutex<Vec<String>>>,
  10 }
  11 impl OpenAiHttp for Replay {
  12     fn post<'a>(
  13         &'a self,
  14         _: &'a str,
  15         _: &'a str,
  16         _: serde_json::Value,
  17     ) -> Pin<Box<dyn Future<Output = Result<String, ToolError>> + Send + 'a>> {
  18         let body = self.responses.lock().unwrap().remove(0);
  19         Box::pin(async move { Ok(body) })
  20     }
  21 }
  22 
  23 #[tokio::main]
  24 async fn main() {
  25     let single = std::env::args().nth(1).as_deref() == Some("single");
  26     let responses = vec![
  27         if single { r#"{"choices":[{"message":{"role":"assistant","content":null,"tool_calls":[{"id":"read-1","type":"function","function":{"name":"read_file","arguments":"{\"path\":\"README.md\"}"}}]}}]}"#.into() } else { r#"{"choices":[{"message":{"role":"assistant","content":null,"tool_calls":[{"id":"search-1","type":"function","function":{"name":"search_files","arguments":"{\"needle\":\"前置\"}"}},{"id":"read-1","type":"function","function":{"name":"read_file","arguments":"{\"path\":\"README.md\"}"}}]}}]}"#.into() },
  28         r#"{"choices":[{"message":{"role":"assistant","content":"我找到了 README.md，并读取了真实内容。"}}]}"#.into(),
  29     ];
  30     let root = std::path::PathBuf::from(env!("CARGO_MANIFEST_DIR"))
  31         .join("../fixtures/workspaces/prerequisites");
  32     let result = run_agent_loop(
  33         &Replay {
  34             responses: Arc::new(Mutex::new(responses)),
  35         },
  36         "offline://replay",
  37         "offline",
  38         "demo",
  39         &Workspace { root },
  40         if single {
  41             "读取 README.md"
  42         } else {
  43             "检查 workspace"
  44         },
  45         32,
  46     )
  47     .await;
  48     println!("{}", serde_json::to_string_pretty(&result).unwrap());
  49 }

  78 impl ReqwestHttp {
  79     pub fn new(timeout: Duration) -> Result<Self, ToolError> {
  80         let client = reqwest::Client::builder()
  81             .timeout(timeout)
  82             .build()
  83             .map_err(|_| ToolError {
  84                 code: "transport_config".into(),
  85                 message: "could not configure HTTP client".into(),
  86             })?;
  87         Ok(Self { client, timeout })
  88     }
  89     pub fn timeout(&self) -> Duration {
  90         self.timeout
  91     }
  92 }

  94 impl OpenAiHttp for ReqwestHttp {
  95     fn post<'a>(
  96         &'a self,
  97         url: &'a str,
  98         api_key: &'a str,
  99         body: serde_json::Value,
 100     ) -> Pin<Box<dyn Future<Output = Result<String, ToolError>> + Send + 'a>> {
 101         Box::pin(async move {
 102             let response = self
 103                 .client
 104                 .post(url)
 105                 .bearer_auth(api_key)
 106                 .json(&body)
 107                 .send()
 108                 .await
 109                 .map_err(|error| ToolError {
 110                     code: if error.is_timeout() {
 111                         "timeout"
 112                     } else {
 113                         "network"
 114                     }
 115                     .into(),
 116                     message: "HTTP request failed".into(),
 117                 })?;
 118             let status = response.status();
 119             let text = response.text().await.map_err(|_| ToolError {
 120                 code: "network".into(),
 121                 message: "could not read HTTP response".into(),
 122             })?;
 123             if !status.is_success() {
 124                 return Err(ToolError {
 125                     code: format!("http_{}", status.as_u16()),
 126                     message: "OpenAI HTTP request failed".into(),
 127                 });
 128             }
 129             Ok(text)
 130         })
 131     }
 132 }

 134 pub async fn openai_complete<H: OpenAiHttp>(
 135     http: &H,
 136     base_url: &str,
 137     api_key: &str,
 138     model: &str,
 139     messages: &[Message],
 140     tools: &[ToolDefinition],
 141 ) -> Result<ModelTurn, ToolError> {
 142     let wire_messages: Vec<_> = messages.iter().map(|message| {
 143         if message.role == "tool" { serde_json::json!({"role":"tool","tool_call_id":message.tool_call_id,"content":message.content}) }
 144         else if message.role == "assistant" && !message.tool_calls.is_empty() { serde_json::json!({"role":"assistant","content":message.content,"tool_calls":message.tool_calls.iter().map(|call| serde_json::json!({"id":call.id,"type":"function","function":{"name":call.name,"arguments":call.arguments.to_string()}})).collect::<Vec<_>>()}) }
 145         else { serde_json::json!({"role":message.role,"content":message.content}) }
 146     }).collect();
 147     let body = serde_json::json!({ "model": model, "messages": wire_messages, "tools": tools.iter().map(|tool| serde_json::json!({"type":"function","function":{"name":tool.name,"description":tool.description,"parameters":tool.input_schema}})).collect::<Vec<_>>() });
 148     let response = http
 149         .post(
 150             &format!("{}/chat/completions", base_url.trim_end_matches('/')),
 151             api_key,
 152             body,
 153         )
 154         .await?;
 155     parse_openai_turn(&response)
 156 }

 158 pub fn parse_openai_turn(body: &str) -> Result<ModelTurn, ToolError> {
 159     let root: serde_json::Value = serde_json::from_str(body).map_err(|_| ToolError {
 160         code: "response_format".into(),
 161         message: "invalid JSON".into(),
 162     })?;
 163     let message = root
 164         .get("choices")
 165         .and_then(|v| v.as_array())
 166         .and_then(|a| a.first())
 167         .and_then(|v| v.get("message"))
 168         .ok_or_else(|| ToolError {
 169             code: "response_format".into(),
 170             message: "missing message".into(),
 171         })?;
 172     let content = match message.get("content") {
 173         None | Some(serde_json::Value::Null) => String::new(),
 174         Some(value) => value
 175             .as_str()
 176             .ok_or_else(|| ToolError {
 177                 code: "response_format".into(),
 178                 message: "message content must be string or null".into(),
 179             })?
 180             .to_owned(),
 181     };
 182     if let Some(value) = message.get("tool_calls") {
 183         if !value.is_array() {
 184             return Err(ToolError {
 185                 code: "response_format".into(),
 186                 message: "tool_calls must be an array".into(),
 187             });
 188         }
 189     }
 190     let mut calls = Vec::new();
 191     for call in message
 192         .get("tool_calls")
 193         .and_then(|v| v.as_array())
 194         .into_iter()
 195         .flatten()
 196     {
 197         let id = call
 198             .get("id")
 199             .and_then(|v| v.as_str())
 200             .filter(|v| !v.is_empty())
 201             .ok_or_else(|| ToolError {
 202                 code: "response_format".into(),
 203                 message: "tool call lacks id".into(),
 204             })?;
 205         let function = call.get("function").ok_or_else(|| ToolError {
 206             code: "response_format".into(),
 207             message: "missing tool function".into(),
 208         })?;
 209         let arguments: serde_json::Value = serde_json::from_str(
 210             function
 211                 .get("arguments")
 212                 .and_then(|v| v.as_str())
 213                 .ok_or_else(|| ToolError {
 214                     code: "arguments_invalid".into(),
 215                     message: "tool arguments must be JSON text".into(),
 216                 })?,
 217         )
 218         .map_err(|_| ToolError {
 219             code: "arguments_invalid".into(),
 220             message: "tool arguments are not JSON".into(),
 221         })?;
 222         if !arguments.is_object() {
 223             return Err(ToolError {
 224                 code: "arguments_invalid".into(),
 225                 message: "tool arguments must be an object".into(),
 226             });
 227         }
 228         let name = function
 229             .get("name")
 230             .and_then(|v| v.as_str())
 231             .filter(|v| !v.is_empty())
 232             .ok_or_else(|| ToolError {
 233                 code: "response_format".into(),
 234                 message: "tool call lacks name".into(),
 235             })?;
 236         calls.push(ToolCall {
 237             id: id.into(),
 238             name: function
 239                 .get("name")
 240                 .and_then(|v| v.as_str())
 241                 .unwrap_or(name)
 242                 .into(),
 243             arguments,
 244         });
 245     }
 246     let message = Message {
 247         role: "assistant".into(),
 248         content,
 249         tool_call_id: None,
 250         tool_calls: calls.clone(),
 251     };
 252     Ok(ModelTurn {
 253         message,
 254         tool_calls: calls,
 255     })
 256 }

 258 pub fn anthropic_request(messages: &[Message], model: &str) -> serde_json::Value {
 259     let system = messages
 260         .iter()
 261         .find(|m| m.role == "system")
 262         .map(|m| m.content.clone());
 263     let mut filtered: Vec<serde_json::Value> = Vec::new();
 264     for m in messages.iter().filter(|m| m.role != "system") {
 265         if m.role == "tool" {
 266             let block = serde_json::json!({"type":"tool_result","tool_use_id":m.tool_call_id,"content":m.content});
 267             if filtered
 268                 .last()
 269                 .and_then(|v| v.get("role"))
 270                 .and_then(|v| v.as_str())
 271                 == Some("user")
 272                 && filtered
 273                     .last()
 274                     .and_then(|v| v.get("content"))
 275                     .and_then(|v| v.as_array())
 276                     .is_some()
 277             {
 278                 filtered.last_mut().unwrap()["content"]
 279                     .as_array_mut()
 280                     .unwrap()
 281                     .push(block);
 282             } else {
 283                 filtered.push(serde_json::json!({"role":"user","content":[block]}));
 284             }
 285         } else if m.role == "assistant" && !m.tool_calls.is_empty() {
 286             let mut content = Vec::new();
 287             if !m.content.is_empty() {
 288                 content.push(serde_json::json!({"type":"text","text":m.content}));
 289             }
 290             content.extend(m.tool_calls.iter().map(|c| serde_json::json!({"type":"tool_use","id":c.id,"name":c.name,"input":c.arguments})));
 291             filtered.push(serde_json::json!({"role":"assistant","content":content}));
 292         } else {
 293             filtered.push(serde_json::json!({"role": if m.role == "assistant" { "assistant" } else { "user" }, "content": m.content}));
 294         }
 295     }
 296     let mut request = serde_json::json!({"model": model, "max_tokens": 1024, "messages": filtered});
 297     if let Some(system) = system {
 298         request["system"] = system.into();
 299     }
 300     request
 301 }

 307 pub fn parse_anthropic_turn(body: &str) -> Result<ModelTurn, ToolError> {
 308     let value: serde_json::Value = serde_json::from_str(body).map_err(|_| ToolError {
 309         code: "response_format".into(),
 310         message: "invalid JSON".into(),
 311     })?;
 312     let blocks = value
 313         .get("content")
 314         .and_then(|v| v.as_array())
 315         .ok_or_else(|| ToolError {
 316             code: "response_format".into(),
 317             message: "missing content".into(),
 318         })?;
 319     let content = blocks
 320         .iter()
 321         .filter(|b| b.get("type").and_then(|t| t.as_str()) == Some("text"))
 322         .filter_map(|b| b.get("text").and_then(|t| t.as_str()))
 323         .collect::<Vec<_>>()
 324         .join("");
 325     let mut calls = Vec::new();
 326     for block in blocks
 327         .iter()
 328         .filter(|b| b.get("type").and_then(|t| t.as_str()) == Some("tool_use"))
 329     {
 330         let id = block
 331             .get("id")
 332             .and_then(|v| v.as_str())
 333             .filter(|s| !s.is_empty())
 334             .ok_or_else(|| ToolError {
 335                 code: "response_format".into(),
 336                 message: "tool_use lacks id".into(),
 337             })?;
 338         let name = block
 339             .get("name")
 340             .and_then(|v| v.as_str())
 341             .filter(|s| !s.is_empty())
 342             .ok_or_else(|| ToolError {
 343                 code: "response_format".into(),
 344                 message: "tool_use lacks name".into(),
 345             })?;
 346         let arguments = block
 347             .get("input")
 348             .filter(|v| v.is_object())
 349             .cloned()
 350             .ok_or_else(|| ToolError {
 351                 code: "arguments_invalid".into(),
 352                 message: "tool input must be an object".into(),
 353             })?;
 354         calls.push(ToolCall {
 355             id: id.into(),
 356             name: name.into(),
 357             arguments,
 358         });
 359     }
 360     for block in blocks {
 361         if block.get("type").and_then(|v| v.as_str()) == Some("text")
 362             && block.get("text").and_then(|v| v.as_str()).is_none()
 363         {
 364             return Err(ToolError {
 365                 code: "response_format".into(),
 366                 message: "text block is invalid".into(),
 367             });
 368         }
 369         if !matches!(
 370             block.get("type").and_then(|v| v.as_str()),
 371             Some("text") | Some("tool_use")
 372         ) {
 373             return Err(ToolError {
 374                 code: "response_format".into(),
 375                 message: "content block is invalid".into(),
 376             });
 377         }
 378     }
 379     if calls.is_empty() && content.is_empty() {
 380         return Err(ToolError {
 381             code: "response_format".into(),
 382             message: "Anthropic response lacks text content".into(),
 383         });
 384     }
 385     Ok(ModelTurn {
 386         message: Message {
 387             role: "assistant".into(),
 388             content,
 389             tool_call_id: None,
 390             tool_calls: calls.clone(),
 391         },
 392         tool_calls: calls,
 393     })
 394 }

 410 fn safe_path(workspace: &Workspace, input: &str) -> Result<PathBuf, ToolError> {
 411     let root = workspace.root.canonicalize().map_err(|_| ToolError {
 412         code: "workspace_invalid".into(),
 413         message: "workspace unavailable".into(),
 414     })?;
 415     let mut components = Vec::new();
 416     for component in Path::new(input).components() {
 417         match component {
 418             std::path::Component::RootDir | std::path::Component::Prefix(_) => {
 419                 return Err(ToolError {
 420                     code: "path_escape".into(),
 421                     message: "path escapes workspace".into(),
 422                 });
 423             }
 424             std::path::Component::ParentDir => {
 425                 if components.pop().is_none() {
 426                     return Err(ToolError {
 427                         code: "path_escape".into(),
 428                         message: "path escapes workspace".into(),
 429                     });
 430                 }
 431             }
 432             std::path::Component::Normal(value) => components.push(value),
 433             std::path::Component::CurDir => {}
 434         }
 435     }
 436     let requested = root.join(input);
 437     let candidate = requested.canonicalize().map_err(|_| ToolError {
 438         code: "path_invalid".into(),
 439         message: "path unavailable".into(),
 440     })?;
 441     if !candidate.starts_with(&root) {
 442         return Err(ToolError {
 443             code: "path_escape".into(),
 444             message: "path escapes workspace".into(),
 445         });
 446     }
 447     Ok(candidate)
 448 }

 450 pub fn read_file(workspace: &Workspace, path: &str) -> ToolResult {
 451     match safe_path(workspace, path).and_then(|path| {
 452         std::fs::read_to_string(path).map_err(|_| ToolError {
 453             code: "read_failed".into(),
 454             message: "read failed".into(),
 455         })
 456     }) {
 457         Ok(output) => ToolResult {
 458             tool_call_id: "read_file".into(),
 459             ok: true,
 460             output: Some(output),
 461             error: None,
 462         },
 463         Err(error) => ToolResult {
 464             tool_call_id: "read_file".into(),
 465             ok: false,
 466             output: None,
 467             error: Some(error),
 468         },
 469     }
 470 }

 472 pub fn search_files(workspace: &Workspace, needle: &str) -> ToolResult {
 473     fn walk(root: &Path, dir: &Path, needle: &str, found: &mut Vec<String>) -> std::io::Result<()> {
 474         for entry in std::fs::read_dir(dir)? {
 475             let entry = entry?;
 476             let name = entry.file_name();
 477             if matches!(name.to_str(), Some(".git" | "target" | "node_modules")) {
 478                 continue;
 479             }
 480             let path = entry.path();
 481             let metadata = std::fs::symlink_metadata(&path)?;
 482             if metadata.file_type().is_symlink() {
 483                 continue;
 484             }
 485             if metadata.is_dir() {
 486                 walk(root, &path, needle, found)?;
 487             } else if path.is_file()
 488                 && std::fs::read_to_string(&path)
 489                     .map(|s| s.contains(needle))
 490                     .unwrap_or(false)
 491             {
 492                 found.push(
 493                     path.strip_prefix(root)
 494                         .unwrap_or(&path)
 495                         .display()
 496                         .to_string(),
 497                 );
 498             }
 499         }
 500         Ok(())
 501     }
 502     let root = match workspace.root.canonicalize() {
 503         Ok(root) => root,
 504         Err(_) => {
 505             return ToolResult {
 506                 tool_call_id: "search_files".into(),
 507                 ok: false,
 508                 output: None,
 509                 error: Some(ToolError {
 510                     code: "workspace_invalid".into(),
 511                     message: "workspace unavailable".into(),
 512                 }),
 513             }
 514         }
 515     };
 516     let mut found = Vec::new();
 517     match walk(&root, &root, needle, &mut found) {
 518         Ok(()) => {
 519             found.sort();
 520             ToolResult {
 521                 tool_call_id: "search_files".into(),
 522                 ok: true,
 523                 output: Some(found.join("\n")),
 524                 error: None,
 525             }
 526         }
 527         Err(_) => ToolResult {
 528             tool_call_id: "search_files".into(),
 529             ok: false,
 530             output: None,
 531             error: Some(ToolError {
 532                 code: "search_failed".into(),
 533                 message: "search failed".into(),
 534             }),
 535         },
 536     }
 537 }

 539 pub fn dispatch_readonly(call: &ToolCall, workspace: &Workspace) -> ToolResult {
 540     match call.name.as_str() {
 541         "read_file" => match call.arguments.get("path").and_then(|v| v.as_str()) {
 542             Some(path) => {
 543                 let mut result = read_file(workspace, path);
 544                 result.tool_call_id = call.id.clone();
 545                 result
 546             }
 547             None => invalid_result(&call.id, "read_file requires string path"),
 548         },
 549         "search_files" => match call.arguments.get("needle").and_then(|v| v.as_str()) {
 550             Some(needle) => {
 551                 let mut result = search_files(workspace, needle);
 552                 result.tool_call_id = call.id.clone();
 553                 result
 554             }
 555             None => invalid_result(&call.id, "search_files requires string needle"),
 556         },
 557         _ => ToolResult {
 558             tool_call_id: call.id.clone(),
 559             ok: false,
 560             output: None,
 561             error: Some(ToolError {
 562                 code: "unknown_tool".into(),
 563                 message: format!("unknown tool: {}", call.name),
 564             }),
 565         },
 566     }
 567 }

 627 pub async fn run_agent_loop<H: OpenAiHttp>(
 628     http: &H,
 629     base_url: &str,
 630     api_key: &str,
 631     model: &str,
 632     workspace: &Workspace,
 633     prompt: &str,
 634     max_turns: usize,
 635 ) -> LoopResult {
 636     let mut messages = vec![Message {
 637         role: "user".into(),
 638         content: prompt.into(),
 639         tool_call_id: None,
 640         tool_calls: vec![],
 641     }];
 642     let mut events = Vec::new();
 643     let definitions = READONLY_TOOLS
 644         .iter()
 645         .map(|(name, description)| ToolDefinition {
 646             name: (*name).into(),
 647             description: (*description).into(),
 648             input_schema: serde_json::json!({"type":"object"}),
 649         })
 650         .collect::<Vec<_>>();
 651     for turn in 1..=max_turns {
 652         events.push(LoopEvent::ModelRequested {
 653             turn,
 654             message_count: messages.len(),
 655         });
 656         let response =
 657             match openai_complete(http, base_url, api_key, model, &messages, &definitions).await {
 658                 Ok(value) => value,
 659                 Err(error) => {
 660                     return finish_loop(
 661                         messages,
 662                         events,
 663                         LoopState::Failed,
 664                         StopReason::ModelError,
 665                         None,
 666                         Some(error.message),
 667                     )
 668                 }
 669             };
 670         let ids = response
 671             .tool_calls
 672             .iter()
 673             .map(|call| call.id.clone())
 674             .collect();
 675         events.push(LoopEvent::ModelReceived {
 676             turn,
 677             tool_call_ids: ids,
 678             text: response.message.content.clone(),
 679         });
 680         messages.push(response.message.clone());
 681         if response.tool_calls.is_empty() {
 682             if response.message.content.trim().is_empty() {
 683                 return finish_loop(
 684                     messages,
 685                     events,
 686                     LoopState::Failed,
 687                     StopReason::EmptyFinal,
 688                     None,
 689                     None,
 690                 );
 691             }
 692             return finish_loop(
 693                 messages,
 694                 events,
 695                 LoopState::Completed,
 696                 StopReason::FinalAnswer,
 697                 Some(response.message.content),
 698                 None,
 699             );
 700         }
 701         for call in response.tool_calls {
 702             let result = dispatch_readonly(&call, workspace);
 703             events.push(LoopEvent::ToolResult {
 704                 turn,
 705                 tool_call_id: call.id.clone(),
 706                 result: result.clone(),
 707             });
 708             let content = if result.ok {
 709                 result.output.unwrap_or_default()
 710             } else {
 711                 serde_json::json!({"ok":false,"error":result.error}).to_string()
 712             };
 713             messages.push(Message {
 714                 role: "tool".into(),
 715                 content,
 716                 tool_call_id: Some(result.tool_call_id),
 717                 tool_calls: vec![],
 718             });
 719         }
 720     }
 721     finish_loop(
 722         messages,
 723         events,
 724         LoopState::Failed,
 725         StopReason::MaxTurns,
 726         None,
 727         None,
 728     )
 729 }
 730 
 731 fn finish_loop(
 732     messages: Vec<Message>,
 733     mut events: Vec<LoopEvent>,
 734     state: LoopState,
 735     reason: StopReason,
 736     answer: Option<String>,
 737     error: Option<String>,
 738 ) -> LoopResult {
 739     events.push(LoopEvent::Stopped {
 740         state: state.clone(),
 741         reason: reason.clone(),
 742     });
 743     LoopResult {
 744         state,
 745         reason,
 746         answer,
 747         messages,
 748         events,
 749         error,
 750     }
 751 }
 752 
 753 fn invalid_result(id: &str, message: &str) -> ToolResult {
 754     ToolResult {
 755         tool_call_id: id.into(),
 756         ok: false,
 757         output: None,
 758         error: Some(ToolError {
 759             code: "arguments_invalid".into(),
 760             message: message.into(),
 761         }),
 762     }
 763 }

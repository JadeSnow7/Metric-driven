   1 import { describe, expect, it } from 'vitest'
   2 import { mkdtemp, writeFile, rm } from 'node:fs/promises'
   3 import { join } from 'node:path'
   4 import { tmpdir } from 'node:os'
   5 import { replayAdapter, runAgentLoop } from '../src/rein/loop'
   6 import type { Message, ModelTurn } from '../src/rein/contracts'
   7 
   8 const assistant = (content: string, toolCalls: ModelTurn['toolCalls'] = []): ModelTurn => ({ message: { role: 'assistant', content, toolCalls }, toolCalls })
   9 
  10 describe('core agent loop', () => {
  11   it('returns a normal final answer', async () => {
  12     const root = await mkdtemp(join(tmpdir(), 'rein-loop-'))
  13     try { const result = await runAgentLoop(replayAdapter([assistant('完成')]), [{ role: 'user', content: 'hi' }], { root }); expect(result).toMatchObject({ ok: true, answer: '完成', reason: 'completed' }); expect(result.events.at(-1)).toEqual({ type: 'stopped', reason: 'completed' }) } finally { await rm(root, { recursive: true, force: true }) }
  14   })
  15 
  16   it('searches and then reads real file contents before answering', async () => {
  17     const root = await mkdtemp(join(tmpdir(), 'rein-loop-'))
  18     try {
  19       await writeFile(join(root, 'notes.txt'), '真实内容：蓝色书签')
  20       const result = await runAgentLoop(replayAdapter([
  21         assistant('', [{ id: 's1', name: 'search_files', arguments: { needle: '蓝色' } }]),
  22         assistant('', [{ id: 'r1', name: 'read_file', arguments: { path: 'notes.txt' } }]),
  23         assistant('蓝色书签'),
  24       ]), [{ role: 'user', content: '找蓝色并总结' }], { root })
  25       expect(result.ok).toBe(true); expect(result.answer).toBe('蓝色书签')
  26       expect(result.messages.some(message => message.content.includes('notes.txt'))).toBe(true)
  27       expect(result.messages.some(message => message.content.includes('真实内容'))).toBe(true)
  28     } finally { await rm(root, { recursive: true, force: true }) }
  29   })
  30 
  31   it('keeps multiple call order and ids, and feeds failures back', async () => {
  32     const root = await mkdtemp(join(tmpdir(), 'rein-loop-'))
  33     try {
  34       await writeFile(join(root, 'a.txt'), 'a')
  35       const result = await runAgentLoop(replayAdapter([
  36         assistant('', [{ id: 'a', name: 'read_file', arguments: { path: 'a.txt' } }, { id: 'b', name: 'no_such_tool', arguments: {} }]),
  37         assistant('已说明失败'),
  38       ]), [{ role: 'user', content: '读文件' }], { root })
  39       expect(result.ok).toBe(true)
  40       expect(result.events.filter(event => event.type === 'tool').map(event => event.toolCallId)).toEqual(['a', 'b'])
  41       expect(result.messages.filter(message => message.role === 'tool').map(message => message.toolCallId)).toEqual(['a', 'b'])
  42       expect(result.messages.at(-1)?.content).toBe('已说明失败')
  43     } finally { await rm(root, { recursive: true, force: true }) }
  44   })
  45 
  46   it('records model errors, empty answers, and max-round protection', async () => {
  47     const root = await mkdtemp(join(tmpdir(), 'rein-loop-'))
  48     try {
  49       await expect(runAgentLoop(replayAdapter([new Error('down')]), [], { root })).resolves.toMatchObject({ ok: false, reason: 'model_error' })
  50       await expect(runAgentLoop(replayAdapter([assistant('')]), [], { root })).resolves.toMatchObject({ ok: false, reason: 'empty_final' })
  51       await expect(runAgentLoop(replayAdapter([assistant('', [{ id: 'x', name: 'nope', arguments: {} }])]), [], { root }, { maxRounds: 1 })).resolves.toMatchObject({ ok: false, reason: 'max_rounds' })
  52     } finally { await rm(root, { recursive: true, force: true }) }
  53   })
  54 })

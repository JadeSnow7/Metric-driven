   1 use rein_ch01_helloworld::rein::{
   2     run_agent_loop, LoopAdapter, Message, ModelTurn, ToolCall, ToolDefinition, ToolError, Workspace,
   3 };
   4 use std::{future::Future, pin::Pin};
   5 struct OfflineModel {
   6     round: usize,
   7 }
   8 impl LoopAdapter for OfflineModel {
   9     fn complete<'a>(
  10         &'a mut self,
  11         messages: &'a [Message],
  12         _tools: &'a [ToolDefinition],
  13     ) -> Pin<Box<dyn Future<Output = Result<ModelTurn, ToolError>> + Send + 'a>> {
  14         self.round += 1;
  15         let make = |id: &str, name: &str, args: serde_json::Value| ToolCall {
  16             id: id.into(),
  17             name: name.into(),
  18             arguments: args,
  19         };
  20         let turn = match self.round {
  21             1 => {
  22                 let calls = vec![make(
  23                     "search-1",
  24                     "search_files",
  25                     serde_json::json!({"needle":"工具结果"}),
  26                 )];
  27                 ModelTurn {
  28                     message: Message {
  29                         role: "assistant".into(),
  30                         content: String::new(),
  31                         tool_call_id: None,
  32                         tool_calls: calls.clone(),
  33                     },
  34                     tool_calls: calls,
  35                 }
  36             }
  37             2 => {
  38                 let calls = vec![make(
  39                     "read-1",
  40                     "read_file",
  41                     serde_json::json!({"path":"guide.md"}),
  42                 )];
  43                 ModelTurn {
  44                     message: Message {
  45                         role: "assistant".into(),
  46                         content: String::new(),
  47                         tool_call_id: None,
  48                         tool_calls: calls.clone(),
  49                     },
  50                     tool_calls: calls,
  51                 }
  52             }
  53             _ => ModelTurn {
  54                 message: Message {
  55                     role: "assistant".into(),
  56                     content: messages
  57                         .iter()
  58                         .find(|m| m.role == "tool")
  59                         .map(|m| format!("读取完成：{}", m.content.trim()))
  60                         .unwrap_or_else(|| "读取完成".into()),
  61                     tool_call_id: None,
  62                     tool_calls: vec![],
  63                 },
  64                 tool_calls: vec![],
  65             },
  66         };
  67         Box::pin(async move { Ok(turn) })
  68     }
  69 }
  70 #[tokio::main]
  71 async fn main() {
  72     let root = std::env::temp_dir().join(format!("rein-ch05-demo-{}", std::process::id()));
  73     let _ = std::fs::remove_dir_all(&root);
  74     std::fs::create_dir_all(&root).unwrap();
  75     std::fs::write(root.join("guide.md"), "# Loop\n工具结果会进入下一轮。\n").unwrap();
  76     let tools = vec![
  77         ToolDefinition {
  78             name: "search_files".into(),
  79             description: "搜索文本文件".into(),
  80             input_schema: serde_json::json!({"type":"object"}),
  81         },
  82         ToolDefinition {
  83             name: "read_file".into(),
  84             description: "读取文本文件".into(),
  85             input_schema: serde_json::json!({"type":"object"}),
  86         },
  87     ];
  88     let mut model = OfflineModel { round: 0 };
  89     let result = run_agent_loop(
  90         &mut model,
  91         &[Message {
  92             role: "user".into(),
  93             content: "搜索并读取 guide.md".into(),
  94             tool_call_id: None,
  95             tool_calls: vec![],
  96         }],
  97         &Workspace { root: root.clone() },
  98         &tools,
  99         32,
 100     )
 101     .await;
 102     println!("{}", serde_json::to_string_pretty(&serde_json::json!({"answer": result.answer, "reason": result.reason, "events": result.events})).unwrap());
 103     let _ = std::fs::remove_dir_all(root);
 104 }

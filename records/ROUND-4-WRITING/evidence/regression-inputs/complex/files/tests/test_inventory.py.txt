from __future__ import annotations

import tempfile
import unittest
from pathlib import Path

from inventory import Inventory


class InventoryTests(unittest.TestCase):
    def test_add_and_count(self) -> None:
        inventory = Inventory()
        inventory.add_item("apple", 3)
        inventory.add_item("apple", 2)
        self.assertEqual(inventory.count("apple"), 5)

    def test_unknown_item_count_is_zero(self) -> None:
        self.assertEqual(Inventory().count("pear"), 0)

    def test_rejects_non_positive_quantity(self) -> None:
        with self.assertRaises(ValueError):
            Inventory().add_item("apple", 0)

    def test_names_sorted(self) -> None:
        inventory = Inventory()
        inventory.add_item("pear", 1)
        inventory.add_item("apple", 1)
        self.assertEqual(inventory.names(), ["apple", "pear"])


class ImportCsvTests(unittest.TestCase):
    def setUp(self) -> None:
        self._tmp = tempfile.TemporaryDirectory()
        self.addCleanup(self._tmp.cleanup)
        self.inventory = Inventory()

    def write(self, content: str | bytes) -> Path:
        path = Path(self._tmp.name) / "stock.csv"
        if isinstance(content, str):
            content = content.encode("utf-8")
        path.write_bytes(content)
        return path

    def test_imports_valid_rows(self) -> None:
        path = self.write("name,quantity\napple,3\npear,2\n")
        self.assertEqual(self.inventory.import_csv(path), (2, []))
        self.assertEqual(self.inventory.count("apple"), 3)
        self.assertEqual(self.inventory.count("pear"), 2)

    def test_accepts_str_path(self) -> None:
        path = self.write("name,quantity\napple,3\n")
        self.assertEqual(self.inventory.import_csv(str(path)), (1, []))

    def test_header_only_imports_nothing(self) -> None:
        path = self.write("name,quantity\n")
        self.assertEqual(self.inventory.import_csv(path), (0, []))
        self.assertEqual(self.inventory.names(), [])

    def test_malformed_rows_are_skipped_by_line_number(self) -> None:
        path = self.write(
            "name,quantity\n"  # 1 header
            "apple,3\n"  # 2 ok
            "banana\n"  # 3 missing field
            "cherry,abc\n"  # 4 not a number
            "date,2,extra\n"  # 5 too many fields
            ",4\n"  # 6 empty name
            "   ,4\n"  # 7 blank name
            "fig,\n"  # 8 empty quantity
            "elder,5\n"  # 9 ok
        )
        self.assertEqual(self.inventory.import_csv(path), (2, [3, 4, 5, 6, 7, 8]))
        self.assertEqual(self.inventory.names(), ["apple", "elder"])
        self.assertEqual(self.inventory.count("elder"), 5)

    def test_quantity_must_be_positive_ascii_integer(self) -> None:
        path = self.write(
            "name,quantity\n"
            "zero,0\n"  # 2
            "negative,-1\n"  # 3
            "decimal,3.0\n"  # 4
            "plus,+3\n"  # 5
            "underscore,1_000\n"  # 6
            "arabic,\u0663\n"  # 7 ARABIC-INDIC DIGIT THREE, int() would accept it
            "padded , 7 \n"  # 8 ok: surrounding whitespace is ignored
            "leading_zero,007\n"  # 9 ok
        )
        self.assertEqual(self.inventory.import_csv(path), (2, [2, 3, 4, 5, 6, 7]))
        self.assertEqual(self.inventory.count("padded"), 7)
        self.assertEqual(self.inventory.count("leading_zero"), 7)
        self.assertEqual(self.inventory.names(), ["leading_zero", "padded"])

    def test_unclosed_quote_only_skips_its_own_line(self) -> None:
        path = self.write('name,quantity\n"apple,3\nbanana,2\ncherry,1\n')
        self.assertEqual(self.inventory.import_csv(path), (2, [2]))
        self.assertEqual(self.inventory.names(), ["banana", "cherry"])

    def test_quoted_fields(self) -> None:
        path = self.write(
            "name,quantity\n"
            '"Widget, large",3\n'  # 2 ok: comma inside quotes
            '"12"" pizza",2\n'  # 3 ok: doubled quote inside quotes
            ' "Bolt" , "4" \n'  # 4 ok: spaces around quoted fields
        )
        self.assertEqual(self.inventory.import_csv(path), (3, []))
        self.assertEqual(self.inventory.count("Widget, large"), 3)
        self.assertEqual(self.inventory.count('12" pizza'), 2)
        self.assertEqual(self.inventory.count("Bolt"), 4)

    def test_stray_quotes_are_malformed(self) -> None:
        path = self.write(
            "name,quantity\n"
            '12" pizza,3\n'  # 2 quote inside an unquoted field
            '"apple"x,3\n'  # 3 text after the closing quote
            '"pe"ar",3\n'  # 4 undoubled quote inside quotes
            "pear,2\n"  # 5 ok
        )
        self.assertEqual(self.inventory.import_csv(path), (1, [2, 3, 4]))
        self.assertEqual(self.inventory.names(), ["pear"])

    def test_quoted_newline_record_is_skipped_entirely(self) -> None:
        # Records are physical lines; neither half of a multi-line record may be imported.
        path = self.write('name,quantity\n"Widget\nlarge",3\npear,2\n')
        self.assertEqual(self.inventory.import_csv(path), (1, [2, 3]))
        self.assertEqual(self.inventory.names(), ["pear"])

    def test_long_values_do_not_affect_other_lines(self) -> None:
        long_name = "n" * 200_000  # longer than the csv module's default 131072-char field limit
        huge_quantity = "9" * 5_000  # above Python's 4300-digit int conversion limit
        path = self.write(
            "name,quantity\n"
            f"{long_name},1\n"  # 2 ok: names have no length limit
            f"apple,{huge_quantity}\n"  # 3
            "pear,2\n"  # 4 ok
        )
        self.assertEqual(self.inventory.import_csv(path), (2, [3]))
        self.assertEqual(self.inventory.count(long_name), 1)
        self.assertEqual(self.inventory.count("pear"), 2)
        self.assertEqual(self.inventory.count("apple"), 0)

    def test_blank_lines_are_ignored_but_keep_line_numbers(self) -> None:
        path = self.write("name,quantity\n\napple,1\n   \nbad\npear,2\n\n")
        self.assertEqual(self.inventory.import_csv(path), (2, [5]))

    def test_crlf_line_endings(self) -> None:
        path = self.write(b"name,quantity\r\napple,1\r\nbad\r\npear,2\r\n")
        self.assertEqual(self.inventory.import_csv(path), (2, [3]))
        self.assertEqual(self.inventory.count("pear"), 2)

    def test_header_with_bom_and_spaces_is_accepted(self) -> None:
        path = self.write("\ufeff name , quantity \napple,1\n".encode("utf-8"))
        self.assertEqual(self.inventory.import_csv(path), (1, []))

    def test_repeated_names_add_up(self) -> None:
        self.inventory.add_item("apple", 1)
        path = self.write("name,quantity\napple,2\napple,3\n")
        self.assertEqual(self.inventory.import_csv(path), (2, []))
        self.assertEqual(self.inventory.count("apple"), 6)

    def test_bad_header_raises_and_imports_nothing(self) -> None:
        cases = {
            "empty file": "",
            "no header": "apple,3\npear,2\n",
            "other names": "item,qty\napple,3\n",
            "wrong order": "quantity,name\n3,apple\n",
            "extra column": "name,quantity,price\napple,3,1\n",
            "different case": "Name,Quantity\napple,3\n",
        }
        for label, content in cases.items():
            with self.subTest(label):
                inventory = Inventory()
                with self.assertRaises(ValueError):
                    inventory.import_csv(self.write(content))
                self.assertEqual(inventory.names(), [])

    def test_undecodable_file_raises_and_imports_nothing(self) -> None:
        path = self.write(b"name,quantity\napple,3\nbad,\xff\n")
        with self.assertRaises(UnicodeDecodeError):
            self.inventory.import_csv(path)
        self.assertEqual(self.inventory.names(), [])

    def test_missing_file_raises(self) -> None:
        with self.assertRaises(FileNotFoundError):
            self.inventory.import_csv(Path(self._tmp.name) / "missing.csv")


class LowStockTests(unittest.TestCase):
    def make(self, **quantities: int) -> Inventory:
        inventory = Inventory()
        for name, quantity in quantities.items():
            inventory.add_item(name, quantity)
        return inventory

    def test_default_threshold_is_five_and_exclusive(self) -> None:
        inventory = self.make(pear=5, fig=4, apple=1, kiwi=6)
        self.assertEqual(inventory.low_stock(), ["apple", "fig"])

    def test_custom_threshold_boundaries(self) -> None:
        inventory = self.make(apple=1, fig=4, pear=5)
        self.assertEqual(inventory.low_stock(threshold=4), ["apple"])
        self.assertEqual(inventory.low_stock(6), ["apple", "fig", "pear"])

    def test_result_is_sorted_regardless_of_insertion_order(self) -> None:
        inventory = self.make(pear=1, banana=1, apple=1)
        self.assertEqual(inventory.low_stock(), ["apple", "banana", "pear"])

    def test_empty_results(self) -> None:
        self.assertEqual(Inventory().low_stock(), [])
        self.assertEqual(self.make(apple=1).low_stock(threshold=1), [])


if __name__ == "__main__":
    unittest.main()

"""Price calculations for the inventory service."""

import math
from decimal import ROUND_HALF_UP, Decimal, localcontext

_CENT = Decimal("0.01")


def apply_discount(price: float, rate: float) -> float:
    """Return the discounted price rounded to whole cents (half a cent rounds up)."""
    if not 0 <= rate <= 1:
        raise ValueError("rate must be between 0 and 1")
    if not math.isfinite(price):
        # inf/nan cannot be quantized; keep the previous float result.
        return price * (1 - rate)
    # Built-in round() rounds half to even and works on the binary float value
    # (2.01 * 0.5 is stored as 1.00499999...), so compute in decimal instead.
    # str() gives the shortest decimal form of the float, e.g. "2.01".
    with localcontext() as ctx:
        # Enough digits for the largest finite float plus two decimal places,
        # so neither the product nor the quantize step loses precision.
        ctx.prec = 400
        discounted = Decimal(str(price)) * (1 - Decimal(str(rate)))
        return float(discounted.quantize(_CENT, rounding=ROUND_HALF_UP))
